\section{Day-1 Quick-Start Onboarding Runbook}

To enable your operational team to commence live operations within 15 minutes of receiving this handover package, complete the following sequential onboarding steps:

\begin{checklistbox}{Day-1 Go-Live Setup (15-Minute Protocol)}
\begin{enumerate}[leftmargin=1.4em, itemsep=4pt]
    \item[\textbf{[ ]}] \textbf{Step 1: First Administrator Authentication}
    \begin{itemize}[leftmargin=1em, itemsep=2pt]
        \item Navigate to the live web application at \texttt{https://cand-w.vercel.app/login}.
        \item Sign in using the master administrator credentials provided in Section 6.
    \end{itemize}
    \item[\textbf{[ ]}] \textbf{Step 2: Password Security Reset}
    \begin{itemize}[leftmargin=1em, itemsep=2pt]
        \item Immediately change the default root password via your profile menu.
    \end{itemize}
    \item[\textbf{[ ]}] \textbf{Step 3: Setup Client TomTom Account \& Update API Key}
    \begin{itemize}[leftmargin=1em, itemsep=2pt]
        \item The developer configured the initial developer account for live traffic telemetry.
        \item Create a permanent corporate account at \texttt{https://developer.tomtom.com}.
        \item Generate your live API key and update \texttt{VITE\_TOMTOM\_API\_KEY} in Vercel.
    \end{itemize}
    \item[\textbf{[ ]}] \textbf{Step 4: Corporate Domain DNS Verification (Resend)}
    \begin{itemize}[leftmargin=1em, itemsep=2pt]
        \item Access \texttt{https://resend.com} and add the 3 DKIM/SPF DNS records to your corporate domain (e.g., \texttt{carbonandwhale.com}).
        \item This ensures all automated client Proof-of-Performance emails deliver directly to inboxes without being marked as spam.
    \end{itemize}
    \item[\textbf{[ ]}] \textbf{Step 5: Provision Staff Accounts by Role}
    \begin{itemize}[leftmargin=1em, itemsep=2pt]
        \item Navigate to \textbf{Admin Console} $\rightarrow$ \textbf{User Management} (\texttt{/admin}).
        \item Create user accounts for your \textbf{Sales Executives}, \textbf{Operations Staff}, and \textbf{Finance Officers}.
    \end{itemize}
    \item[\textbf{[ ]}] \textbf{Step 6: Review Annual Regional Holiday Calendar}
    \begin{itemize}[leftmargin=1em, itemsep=2pt]
        \item Go to \textbf{Admin Console} $\rightarrow$ \textbf{Settings} and verify regional Kerala state holidays (\textit{Onam, Vishu, Eid, Christmas}).
        \item The automated 5-day SLA queue engine uses these dates to accurately calculate business-day hold periods.
    \end{itemize}
    \item[\textbf{[ ]}] \textbf{Step 7: Install Progressive Web App (PWA) on Mobile Devices}
    \begin{itemize}[leftmargin=1em, itemsep=2pt]
        \item Have field inspectors open the site on mobile (Chrome on Android or Safari on iOS) and tap \textbf{Add to Home Screen} to enable offline GPS camera capabilities.
    \end{itemize}
\end{enumerate}
\end{checklistbox}

\section{Role-Based User Manual \& Standard Operating Procedures}

\subsection{Administrator (Admin) Manual \& SOPs}

\begin{rolebox}{Administrator}{Admin}
\textbf{Primary Responsibilities}: User lifecycle governance, global SLA parameters, regional holiday calendars, system pricing, and forensic audit log review.
\end{rolebox}

\begin{sopbox}{Admin SOP 1: User Account Provisioning and Role Elevation}
\begin{enumerate}[leftmargin=1.2em, itemsep=2pt]
    \item Navigate to the \textbf{Admin Console} from the sidebar (\texttt{/admin}).
    \item Select the \textbf{User Management} tab.
    \item Click \textbf{Invite New User}. Enter the staff member's name, corporate email address, and temporary password.
    \item Assign the appropriate role: \textit{Admin}, \textit{Sales}, \textit{Ops}, or \textit{Finance}.
    \item Click \textbf{Create User}. The user can immediately log in at \texttt{/login}.
    \item \textbf{Role Modification}: To adjust permissions, click the user row, select the dropdown, and choose the updated role.
\end{enumerate}
\end{sopbox}

\begin{sopbox}{Admin SOP 2: Configuring Global SLA Timers and Kerala Holidays}
\begin{enumerate}[leftmargin=1.2em, itemsep=2pt]
    \item Open \textbf{Admin Console} $\rightarrow$ \textbf{System Settings}.
    \item Under \textbf{Queue SLA Configuration}, set the \textit{Active Queue Business Days} (default: 5 business days).
    \item Under \textbf{Kerala State Holidays}, review active state holidays. Click \textbf{+ Add Holiday Date} to register new regional dates.
    \item The automated background SLA engine immediately incorporates these dates into all hold calculations.
\end{enumerate}
\end{sopbox}

\begin{sopbox}{Admin SOP 3: Reviewing System Audit Logs and CSV Export}
\begin{enumerate}[leftmargin=1.2em, itemsep=2pt]
    \item Navigate to \textbf{Audit Trail} (\texttt{/audit}).
    \item Filter records by entity type (\textit{Asset, Campaign, Queue Entry, User, Settings}) or by specific employee.
    \item Click the upward \textbf{Export} button to download the entire immutable audit log as a CSV spreadsheet.
\end{enumerate}
\end{sopbox}

\newpage

\subsection{Sales Executive (Sales) Manual \& SOPs}

\begin{rolebox}{Sales Executive}{Sales}
\textbf{Primary Responsibilities}: Managing customer brands, checking screen availability, negotiating campaign durations, and booking queue slots.
\end{rolebox}

\begin{sopbox}{Sales SOP 1: Registering Client Brands and Batch CSV Import}
\begin{enumerate}[leftmargin=1.2em, itemsep=2pt]
    \item Navigate to \textbf{Brands} (\texttt{/brands}) and click \textbf{+ New brand}.
    \item Enter the mandatory fields:
    \begin{itemize}[leftmargin=1em, itemsep=2pt]
        \item \textbf{Brand Name}: Official corporate name (e.g., \textit{Tata Motors}).
        \item \textbf{Contact Email} (\texttt{contact\_email}): Primary marketing recipient for automated Proof-of-Performance emails (e.g., \texttt{south.media@tatamotors.com}).
        \item \textbf{Contact Person \& Phone}: Direct client contact point.
        \item \textbf{Industry}: Retail, FMCG, Automotive, Healthcare, etc.
    \end{itemize}
    \item Click \textbf{Save Brand}. A real-time toast alert is broadcast across all staff.
    \item \textbf{Bulk Import}: Click \textbf{Import CSV} and upload a spreadsheet with columns \texttt{name, contact\_email, contact\_person, contact\_phone, industry, notes}.
\end{enumerate}
\end{sopbox}

\begin{sopbox}{Sales SOP 2: Checking Availability and Submitting Queue Reservations}
\begin{enumerate}[leftmargin=1.2em, itemsep=2pt]
    \item Open \textbf{Assets} (\texttt{/assets}). Filter by district (e.g., \textit{Ernakulam}) or benchmark mall (e.g., \textit{Nexus Mall Kochi}).
    \item Check the status pin: \textit{Available} (green) or \textit{In Queue} (amber).
    \item Click \textbf{Reserve slot} on the desired display.
    \item Select the brand, specify the proposed flight duration (e.g., 30 days), and choose proposed start/end dates.
    \item Submit the reservation:
    \begin{itemize}[leftmargin=1em, itemsep=2pt]
        \item If the asset is \textit{Available}, your brand receives the \textbf{Active Slot} with a 5-business-day hold SLA.
        \item If already occupied, your reservation joins the \textbf{Waitlist} at position \texttt{\#1}, \texttt{\#2}, etc.
    \end{itemize}
\end{enumerate}
\end{sopbox}

\newpage

\subsection{Operations \& Field Inspector (Ops) Manual \& SOPs}

\begin{rolebox}{Operations Team}{Ops}
\textbf{Primary Responsibilities}: Checklist execution, anti-fraud Geo-Tagged Proof (GTP) mobile capture, and site inspection management.
\end{rolebox}

\begin{sopbox}{Ops SOP 1: Completing Display Onboarding Checklists}
\begin{enumerate}[leftmargin=1.2em, itemsep=2pt]
    \item Open the campaign in \textbf{Campaign Detail} (\texttt{/campaigns/:id}).
    \item Under the \textbf{Checklist} tab, execute and verify the four mandatory milestones:
    \begin{itemize}[leftmargin=1em, itemsep=2pt]
        \item \textbf{Creative Brief}: Upload approved high-res artwork.
        \item \textbf{Site Inspection}: Verify structural flex mounting area and power supply.
        \item \textbf{Printing \& Dispatch}: Log flex printing completion.
        \item \textbf{Mounting \& Installation}: Verify on-site mounting or digital loop playback.
    \end{itemize}
    \item Mark all four items as \textit{Done} and click \textbf{Mark Onboarded}. The campaign advances to \textbf{Invoicing}.
\end{enumerate}
\end{sopbox}

\begin{sopbox}{Ops SOP 2: Capturing Anti-Fraud GTP Photos in the Field}
\begin{enumerate}[leftmargin=1.2em, itemsep=2pt]
    \item Open the web application or installed PWA on your smartphone.
    \item Navigate to the campaign $\rightarrow$ \textbf{GTP Proofs} tab $\rightarrow$ Tap \textbf{Capture Proof}.
    \item Select the native camera.
    \item \textbf{Geofence Validation}: The system records live device GPS coordinates, timestamp, and device orientation, and verifies that you are physically within the registered 500m geofence radius of the display.
    \item Tap \textbf{Submit Proof}. The photo is compressed on-device and stored in the cloud. It immediately reflects on the client's live portal.
\end{enumerate}
\end{sopbox}

\newpage

\subsection{Finance Officer (Finance) Manual \& SOPs}

\begin{rolebox}{Finance Officer}{Finance}
\textbf{Primary Responsibilities}: Reviewing booking documents, generating 18\% GST tax invoices, recording payments, and activating live display flights.
\end{rolebox}

\begin{sopbox}{Finance SOP 1: Reviewing and Confirming Queue Bookings}
\begin{enumerate}[leftmargin=1.2em, itemsep=2pt]
    \item Open the \textbf{Queue} ledger (\texttt{/queue}).
    \item Locate the active reservation and click \textbf{Confirm}.
    \item Select the \textbf{Confirmation Basis} (\textit{Advance Payment Received}, \textit{Purchase Order Executed}, or \textit{Signed MSA}) and attach the supporting PDF receipt/document.
    \item Confirm the agreed duration in days and click \textbf{Confirm Interest}.
    \item The system moves the asset to \textbf{Onboarding} and auto-cancels competing waitlist entries.
\end{enumerate}
\end{sopbox}

\begin{sopbox}{Finance SOP 2: Raising 18\% GST Invoices and Activating Live Campaign}
\begin{enumerate}[leftmargin=1.2em, itemsep=2pt]
    \item Once Onboarding is completed, open the campaign in the \textbf{Invoicing} stage.
    \item Click \textbf{Raise GST Invoice}.
    \item Enter the invoice ID (e.g., \texttt{INV-2026-0042}) and base taxable amount. Standard 18\% GST is calculated automatically.
    \item Upload the tax invoice PDF and click \textbf{Record Invoice}.
    \item The campaign status immediately becomes \textbf{Live}, screen statuses update, and an automated notification email is sent to the client.
\end{enumerate}
\end{sopbox}
